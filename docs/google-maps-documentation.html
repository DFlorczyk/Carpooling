🗺️ Przegląd architektury

Aplikacja carpooling używa Google Maps JavaScript API v3 do interaktywnej wizualizacji tras przejazdu na stronie głównej (home.html). Mapa jest inicjalizowana w przeglądarce — cały kod rysowania tras działa po stronie klienta. Backend udostępnia dane przez REST API.

Trzy strony używają Google Maps: home.html (Available Rides) · create-ride.html (tworzenie przejazdu) · ride-details.html (szczegóły)
Ta dokumentacja skupia się na home.html — jest to najważniejsza i najbardziej złożona implementacja.

function initMap() {
    // Domyślne centrum mapy: Warszawa (Centrum)
    const defaultLoc = { lat: 52.2297, lng: 21.0122 };

    // Tworzy obiekt mapy Google i osadza go w elemencie #map
    map = new google.maps.Map(document.getElementById("map"), {
        center: defaultLoc,
        zoom: 12,
        styles: STYLES_3   // styl domyślny dla zoom 10–13

    // (applyStep !== null). Obsługuje wybór punktów pickup i dropoff.
    map.addListener("click", function(event) {
        if (!applyStep) return;  // ignoruj kliknięcia poza trybem aplikowania
        const lat = event.latLng.lat();
        const lng = event.latLng.lng();
        // ... (szczegóły w sekcji "Aplikowanie")
    });

    // Serwis wyznaczania tras – jeden instancja na całą stronę
    directionsService = new google.maps.DirectionsService();

    // Pobierz i wyrenderuj listę dostępnych przejazdów
    loadRides();
}

function applyStylesForZoom(zoom) {

function loadRides() {
    // Odczytaj wartości filtrów z formularza HTML
    const dateFrom = document.getElementById("filterDateFrom").value;
    const dateTo   = document.getElementById("filterDateTo").value;
    const price    = document.getElementById("filterPrice").value;

    // Zbuduj parametry zapytania – tylko te, które użytkownik wypełnił
    const params = new URLSearchParams();
    if (dateFrom) params.append("dateFrom", dateFrom + ":00");
    if (dateTo)   params.append("dateTo",   dateTo   + ":00");
    if (price)    params.append("maxPrice", price);

    // GET /api/r/rides/not-started?dateFrom=...&dateTo=...&maxPrice=...
    fetch("/api/r/rides/not-started?" + params.toString())
        .then(res => res.json())
        .then(data => renderRides(data))   // przekaż tablicę RideDTO do renderowania
        .catch(err => console.error("ERROR:", err));
}

public List<RideDTO> getFilteredRides(
        LocalDateTime dateFrom, LocalDateTime dateTo, Double maxPrice) {

    // Jeśli żaden filtr nie jest podany → zwróć wszystkie nierozpoczęte
    if (dateFrom == null && dateTo == null && maxPrice == null) {
        return rideRepository.findDTOsByStateName("not started");
    }
    // Ustaw wartości domyślne dla brakujących filtrów
    if (dateFrom == null) dateFrom = LocalDateTime.of(2000, 1, 1, 0, 0);
    if (dateTo   == null) dateTo   = LocalDateTime.of(2100, 1, 1, 0, 0);
    if (maxPrice == null) maxPrice = Double.MAX_VALUE;
    return rideRepository.findFilteredRidesWithAllParams(dateFrom, dateTo, maxPrice);
}

🃏 Renderowanie kart przejazdów

home.html · renderRides() · linie 501–556

Funkcja renderRides() tworzy dynamicznie karty HTML dla każdego przejazdu. Kluczowe: adresy wyświetlane na kartach są uzyskiwane przez reverse geocoding — współrzędne GPS (lat, lng) są zamieniane na czytelne nazwy dzielnic.

home.html · renderRides() · linie 501–556 (skrócone) JavaScript 
function renderRides(rides) {
    const container = document.getElementById("ridesList");
    container.innerHTML = "";

    rides.forEach(async r => {
        const el = document.createElement("div");
        el.className = "ride-card";

        // Tworzy kartę z placeholderem "Loading..." dla adresu
        el.innerHTML = `
            <div class="ride-route">Loading...</div>
            <div class="ride-meta">
                <span>${r.driverName}</span>
                <span>${r.cost} zł</span>
                <span>${formattedDate}</span>
            </div>
            <button class="details-btn" ...>Details</button>
            ${!isOwnRide ? `<button class="apply-btn" onclick="openApplyPanel(${r.id})">Apply</button>` : ''}
        `;

        // ★ KLUCZOWE: listener kliknięcia całej karty → rysuje trasę na mapie
        el.addEventListener("click", () => {
            showRoute(
                r.id,
                { lat: r.startLatitude,  lng: r.startLongitude },
                { lat: r.endLatitude,    lng: r.endLongitude }
            );
        });

        container.appendChild(el);

        // ASYNC: pobierz czytelne nazwy dla start i koniec równolegle
        const [startName, endName] = await Promise.all([
            reverseGeocodeLatLng(r.startLatitude,  r.startLongitude),
            reverseGeocodeLatLng(r.endLatitude,    r.endLongitude)
        ]);

        // Zastąp "Loading..." prawdziwym adresem
        el.querySelector(".ride-route").textContent = `${startName} → ${endName}`;
    });
}
Asynchroniczne ładowanie adresów Karty są dodawane do DOM natychmiast (z "Loading..."), a adresy wypełniają się asynchronicznie po powrocie odpowiedzi z Geocoder API. Dzięki temu interfejs nie blokuje się na czekanie.

★ Kliknięcie przejazdu — pełny flow

To jest najważniejszy element całej funkcjonalności. Po kliknięciu karty przejazdu uruchamia się łańcuch operacji: odczyt waypointów z serwera → zapytanie do DirectionsService → rysowanie kolorowych polilinii i markerów.

1
Frontend Kliknięcie karty przejazdu Event listener na elemencie .ride-card wywołuje showRoute(r.id, startLatLng, endLatLng). Przekazuje ID przejazdu oraz współrzędne punktu startowego i końcowego.
2
Frontend showRoute() — wyzerowanie i token Inkrementuje _routeToken (zabezpieczenie przed race condition), wywołuje clearRoute() usuwając poprzednią trasę z mapy, a następnie robi fetch do backendu po waypointy przejazdu.
3
Backend GET /api/r/rides/{id}/waypoints Serwer zwraca tablicę obiektów WaypointDTO — punkty pośrednie (przystanki) przypisane do tego przejazdu, posortowane według kolejności (sequenceOrder). Każdy waypoint ma typ: PICKUP lub DROPOFF.
4
Google API DirectionsService.route() — wyznaczanie trasy Waypointy są przekazywane do DirectionsService z flagą optimizeWaypoints: true. Google API oblicza optymalną kolejność przystanków i zwraca trasę podzieloną na legs (odcinki), każdy z listą kroków i ścieżką GPS.
5
Frontend drawRoute() — rysowanie kolorowych segmentów Dla każdego odcinka (leg) tworzona jest osobna google.maps.Polyline z kolorem zależnym od tego, dokąd jedzie ten odcinek: zielony → PICKUP, pomarańczowy → DROPOFF, niebieski → koniec trasy.
6
Frontend placeMarker() — markery na mapie Ustawiane są markery SVG: szary S (start), zielone P (pickup), pomarańczowe D (dropoff), niebieski E (end). Mapa automatycznie dopasowuje zoom i przesunięcie do granic trasy (map.fitBounds).

🔄 showRoute() — szczegółowy opis

home.html · linie 643–675
home.html · showRoute() · linie 643–675 JavaScript 
// Zmienne stanu trasy na poziomie modułu
let _legPolylines = [];   // tablica aktywnych Polyline na mapie
let _customMarkers = [];  // tablica aktywnych Marker na mapie
let _routeToken = 0;       // licznik zapobiegający race condition

function showRoute(rideId, startLatLng, endLatLng) {
    if (!map || !directionsService) return;

    // ① Race-condition guard: każde kliknięcie dostaje unikalny token.
    //    Jeśli użytkownik kliknie szybko inny przejazd przed odpowiedzią
    //    serwera, stary callback sprawdzi token i zrezygnuje z rysowania.
    const token = ++_routeToken;

    // ② Usuń poprzednią trasę z mapy natychmiast
    clearRoute();

    // ③ Pobierz punkty pośrednie (waypoints) z backendu
    fetch(`/api/r/rides/${rideId}/waypoints`)
        .then(res => res.json())
        .then(waypoints => {
            // Sprawdź token – czy to nadal aktualne kliknięcie?
            if (token !== _routeToken) return;

            // ④ Mapuj WaypointDTO na format wymagany przez DirectionsService
            const stops = waypoints.map(wp => ({
                location: { lat: wp.latitude, lng: wp.longitude },
                stopover: true  // stopover:true = liczy jako odrębny odcinek trasy
            }));

            // ⑤ Zapytaj Google DirectionsService o trasę drogową
            directionsService.route(
                {
                    origin:            startLatLng,      // punkt startowy przejazdu
                    destination:       endLatLng,        // punkt końcowy przejazdu
                    waypoints:         stops,            // przystanki pośrednie
                    optimizeWaypoints: true,           // Google optymalizuje kolejność
                    travelMode: google.maps.TravelMode.DRIVING
                },
                (result, status) => {
                    if (token !== _routeToken) return; // kolejne sprawdzenie tokenu
                    if (status === 'OK') {
                        drawRoute(result, waypoints, startLatLng, endLatLng);
                    } else {
                        console.error('Directions request failed:', status);
                    }
                }
            );
        })
        .catch(err => console.error('Failed to load waypoints:', err));
}
Dlaczego _routeToken? Jeśli użytkownik kliknie szybko dwa różne przejazdy, oba wywołają fetch i DirectionsService.route(). Bez tokenu — whichever request zwróci się ostatni, ta trasa zostanie narysowana — co może być stara. Token gwarantuje, że na mapie zawsze widać trasę ostatniego klikniętego przejazdu.
optimizeWaypoints: true Gdy przejazd ma wiele przystanków, Google Maps może zmienić ich kolejność na optymalną trasę. Zwraca pole waypoint_order w odpowiedzi — tablica indeksów wskazująca nową kolejność. drawRoute() używa jej do prawidłowego przypisania kolorów segmentów.

🎨 drawRoute() — rysowanie trasy

home.html · linie 596–641

drawRoute() otrzymuje obiekt DirectionsResult z Google Maps API i rysuje każdy odcinek trasy jako osobną kolorową polilinię. Dzięki temu ta sama trasa może mieć segmenty w różnych kolorach (zielony do pickupu, pomarańczowy do dropoffu, niebieski do celu).

home.html · drawRoute() · linie 596–641 JavaScript 
function drawRoute(result, originalWaypoints, startLatLng, endLatLng) {
    const route = result.routes[0];

    // ① Obsługa optimizeWaypoints – wyznacz rzeczywistą kolejność waypointów
    //    waypoint_order to np. [2,0,1] gdy Google zoptymalizował trasę.
    //    Jeśli brak waypointów, utwórz identyczne mapowanie [0,1,2,...]
    const order = (route.waypoint_order && route.waypoint_order.length > 0)
        ? route.waypoint_order
        : originalWaypoints.map((_, i) => i);

    // Przeindeksuj oryginalne waypointy zgodnie z kolejnością zoptymalizowaną
    const orderedWaypoints = order.map(idx => originalWaypoints[idx]);

    // ② Iteruj przez legs – każdy leg to odcinek między dwoma punktami trasy
    route.legs.forEach((leg, i) => {

        // ③ Dobierz kolor segmentu na podstawie celu tego odcinka:
        //    - i < orderedWaypoints.length → jedzie do waypointa pośredniego
        //      · PICKUP  → zielony  (#22c55e)
        //      · DROPOFF → pomarańczowy (#f97316)
        //    - i === orderedWaypoints.length → jedzie do końca → niebieski
        const color = i < orderedWaypoints.length
            ? (orderedWaypoints[i].type === 'PICKUP' ? '#22c55e' : '#f97316')
            : '#3b82f6';

        // ④ Złóż pełną ścieżkę GPS z kroków (steps) danego odcinka
        const path = [];
        leg.steps.forEach(step => {
            if (step.path) step.path.forEach(pt => path.push(pt));
        });

        // ⑤ Utwórz Polyline i dodaj ją do mapy
        if (path.length > 0) {
            _legPolylines.push(new google.maps.Polyline({
                path,
                strokeColor:   color,
                strokeWeight:  5,      // grubość linii w pikselach
                strokeOpacity: 0.85,   // lekka przezroczystość
                map                    // przypisanie do aktywnej mapy
            }));
        }
    });

    // ⑥ Postaw markery na kluczowych punktach trasy
    placeMarker(startLatLng, '#475569', 'S', 'Start');
    orderedWaypoints.forEach(wp => {
        const isPickup = wp.type === 'PICKUP';
        placeMarker(
            { lat: wp.latitude, lng: wp.longitude },
            isPickup ? '#22c55e' : '#f97316',
            isPickup ? 'P' : 'D',
            isPickup ? 'Pickup' : 'Dropoff'
        );
    });
    placeMarker(endLatLng, '#3b82f6', 'E', 'End');

    // ⑦ Dopasuj widok mapy do granic całej trasy (z marginesem 40px)
    map.fitBounds(route.bounds, { top: 40, right: 40, bottom: 40, left: 40 });
}

Jak działają legs i waypoint_order

Google DirectionsService dzieli trasę na legs:

  • leg[0] → od startu do pierwszego waypointa
  • leg[1] → od pierwszego waypointa do drugiego
  • leg[n] → od ostatniego waypointa do końca

Gdy optimizeWaypoints: true, Google może zmienić kolejność waypointów. Pole route.waypoint_order to tablica np. [2, 0, 1] — oznacza, że w zoptymalizowanej trasie: leg[0] jedzie do oryginalnego waypointa nr 2, leg[1] do nr 0, leg[2] do nr 1. Kod używa tego do prawidłowego przypisania kolorów.

function makeMarkerIcon(color, label) {
    // Dopasuj rozmiar czcionki i szerokość w zależności od długości etykiety
    const fontSize = label.length <= 2 ? 12 : 9;
    const charW    = label.length <= 2 ? 8  : 6;
    const w  = Math.max(26, label.length * charW + 16); // min. szerokość 26px
    const h  = 26;
    const cx = w / 2; // środek poziomy

    const svg = `<svg xmlns="http://www.w3.org/2000/svg" width="${w}" height="${h + 8}">
        <rect x="0" y="0" width="${w}" height="${h}" rx="6" fill="${color}"/>
        <polygon points="${cx-5},${h} ${cx+5},${h} ${cx},${h+8}" fill="${color}"/>
        <text x="${cx}" y="${h/2 + fontSize/3 + 1}" text-anchor="middle"
              fill="white" font-size="${fontSize}"
              font-family="Arial,sans-serif" font-weight="bold">${label}</text>
    </svg>`;

    return {
        url: 'data:image/svg+xml;charset=UTF-8,' + encodeURIComponent(svg),
        scaledSize: new google.maps.Size(w, h + 8),
        anchor: new google.maps.Point(cx, h + 8)  // punkt zakotwiczenia = koniec wskaźnika
    };
}

function placeMarker(position, color, label, title) {
    _customMarkers.push(new google.maps.Marker({
        position, map,
        icon: makeMarkerIcon(color, label),
        title  // tooltip przy najechaniu
    }));
}
Dlaczego anchor: Point(cx, h+8)? Google Maps musi wiedzieć, który piksel ikony odpowiada dokładnej lokalizacji geograficznej. Ustawiamy go na koniec trójkątnego wskaźnika (dolny wierzchołek), żeby marker precyzyjnie wskazywał na mapę — nie na środek prostokąta.

public List<WaypointDTO> getWaypointsForRide(Long rideId) {
    return rideWaypointRepository
        // Pobierz z bazy, posortowane po sequenceOrder rosnąco
        .findByRideIdOrderBySequenceOrderAsc(rideId)
        .stream()
        .map(w -> new WaypointDTO(
            w.getLatitude(),
            w.getLongitude(),
            w.getType().name(),       // "PICKUP" lub "DROPOFF"
            w.getSequenceOrder()
        ))
        .toList();
}

@Transactional
public void applyForRide(Long rideId, ApplyRideForm form, UserProfile passenger) {
    Ride ride = rideRepository.findById(rideId)
        .orElseThrow(() -> new RuntimeException("Ride not found"));

    // Walidacja 1: Kierowca nie może aplikować na własny przejazd
    boolean isDriver = ride.getParticipants().stream()
        .anyMatch(p -> "driver".equalsIgnoreCase(p.getRole().getName())
                    && p.getUser().getId().equals(passenger.getId()));
    if (isDriver) throw new RuntimeException("You cannot apply for your own ride");

    // Walidacja 2: Użytkownik nie może aplikować dwa razy
    rideRequestRepository.findByUser_IdAndRide_Id(passenger.getId(), rideId)
        .ifPresent(r -> { throw new RuntimeException("You have already applied for this ride"); });

    // Zapisz zgłoszenie z współrzędnymi pickup/dropoff wybranymi na mapie
    RideRequest request = new RideRequest();
    request.setRide(ride);
    request.setUser(passenger);
    request.setIsAccepted(null);  // null = oczekuje na decyzję kierowcy
    request.setPickupLatitude(form.getPickupLatitude());
    request.setPickupLongitude(form.getPickupLongitude());
    request.setDropoffLatitude(form.getDropoffLatitude());
    request.setDropoffLongitude(form.getDropoffLongitude());
    rideRequestRepository.save(request);
}

function reverseGeocodeLatLng(lat, lng) {
    return new Promise((resolve) => {
        const geocoder = new google.maps.Geocoder();
        geocoder.geocode({ location: { lat, lng } }, (results, status) => {
            if (status === "OK" && results[0]) {
                const components = results[0].address_components;

                // Priorytet: dzielnica > powiat > miasto > pełny adres
                // Zwracamy najbardziej lokalną nazwę, nie cały adres ulicy
                const neighbourhood = components.find(c =>
                    c.types.includes("neighbourhood") ||
                    c.types.includes("sublocality"));
                const district = components.find(c =>
                    c.types.includes("administrative_area_level_2"));
                const city = components.find(c =>
                    c.types.includes("locality"));

                resolve(
                    neighbourhood?.long_name ||
                    district?.long_name     ||
                    city?.long_name         ||
                    results[0].formatted_address  // fallback: pełny adres
                );
            } else {
                resolve(`${lat}, ${lng}`);  // fallback: surowe współrzędne
            }
        });
    });
}
Priorytet nazwy miejsca Funkcja zwraca jak najbardziej lokalną i krótką nazwę: najpierw szuka nazwy dzielnicy (neighbourhood), potem powiatu (administrative_area_level_2), potem nazwy miasta (locality), a dopiero na końcu używa pełnego adresu formatowanego. Dzięki temu karta przejazdu pokazuje np. "Mokotów → Wola" zamiast pełnych adresów ulic.
